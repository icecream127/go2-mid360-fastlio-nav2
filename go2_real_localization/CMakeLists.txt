cmake_minimum_required(VERSION 3.16)
project(go2_real_localization)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters registration kdtree)
find_package(fast_gicp REQUIRED)
find_package(Threads REQUIRED)
add_library(localization_core STATIC src/localization_state.cpp src/registration_pipeline.cpp)
target_include_directories(localization_core PUBLIC include ${PCL_INCLUDE_DIRS})
target_link_libraries(localization_core ${PCL_LIBRARIES})
ament_target_dependencies(localization_core fast_gicp)
add_executable(gicp_localizer src/main.cpp src/gicp_localizer_node.cpp)
target_include_directories(gicp_localizer PRIVATE include ${PCL_INCLUDE_DIRS})
target_link_libraries(gicp_localizer localization_core Threads::Threads)
ament_target_dependencies(gicp_localizer rclcpp rcl_interfaces sensor_msgs nav_msgs geometry_msgs std_msgs message_filters tf2_ros pcl_conversions fast_gicp)
install(TARGETS gicp_localizer DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS tools/pcd_to_nav2_map.py DESTINATION lib/${PROJECT_NAME} RENAME pcd_to_nav2_map)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_localization_core test/test_localization_core.cpp)
  target_link_libraries(test_localization_core localization_core Threads::Threads)
endif()
ament_package()
